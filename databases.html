CREATE DATABASE IF NOT EXISTS SUMMER;
USE SUMMER;

CREATE TABLE IF NOT EXISTS ΙΣ (
    κωδι    INT PRIMARY KEY,
    ονομα   VARCHAR(255),

Ζεύξεις πινάκων

1 Ποιοι κράτησαν κόκκινο σκάφος — ονόματα & επίπεδα
SELECT Ι.ονομα, Ι.επίπεδο
FROM ΙΣ Ι, ΚΡ Κ, ΣΚ Σ
WHERE Ι.κωδι = Κ.κωδι
  AND Κ.κωδσ = Σ.κωδσ
  AND Σ.χρωμα = 'Κόκκινο';
2 Όλοι όσοι έχουν κράτηση — αλφαβητικά
SELECT Ι.ονομα
FROM ΙΣ Ι, ΚΡ Κ
WHERE Ι.κωδι = Κ.κωδι
ORDER BY ονομα;
3 Κράτηση σε κόκκινο ή πράσινο σκάφος — OR
SELECT Ι.ονομα
FROM ΙΣ Ι, ΚΡ Κ, ΣΚ Σ
WHERE Ι.κωδι = Κ.κωδι
  AND Σ.κωδσ = Κ.κωδσ
  AND (Σ.χρωμα = 'Κόκκινο' OR Σ.χρωμα = 'Πράσινο');
4 Κράτηση και σε κόκκινο και σε πράσινο — self-ζεύξη
SELECT Ι.ονομα
FROM ΙΣ Ι, ΚΡ Κ1, ΚΡ Κ2, ΣΚ Σ1, ΣΚ Σ2
WHERE Ι.κωδι = Κ1.κωδι AND Σ1.κωδσ = Κ1.κωδσ AND Σ1.χρωμα = 'Κόκκινο'
  AND Ι.κωδι = Κ2.κωδι AND Σ2.κωδσ = Κ2.κωδσ AND Σ2.χρωμα = 'Πράσινο';
5 Δύο διαφορετικά σκάφη την ίδια ημέρα
SELECT DISTINCT Ι.ονομα
FROM ΙΣ Ι, ΚΡ Κ1, ΚΡ Κ2
WHERE Ι.κωδι = Κ1.κωδι
  AND Ι.κωδι = Κ2.κωδι
  AND Κ1.ημ = Κ2.ημ
  AND Κ1.κωδσ <> Κ2.κωδσ;
6 Δύο σκάφη ίδια μέρα — με bonus +2 στο επίπεδο
SELECT DISTINCT Ι.ονομα, Ι.επίπεδο + 2
FROM ΙΣ Ι, ΚΡ Κ1, ΚΡ Κ2
WHERE Ι.κωδι = Κ1.κωδι
  AND Ι.κωδι = Κ2.κωδι
  AND Κ1.ημ = Κ2.ημ
  AND Κ1.κωδσ <> Κ2.κωδσ;
7 Κράτηση σε συγκεκριμένο σκάφος (κωδ. 2)
SELECT DISTINCT Ι.ονομα
FROM ΙΣ Ι, ΚΡ Κ, ΣΚ Σ
WHERE Ι.κωδι = Κ.κωδι
  AND Κ.κωδσ = 2;
8 Τι χρώμα σκάφη κράτησε η Μαρία
SELECT DISTINCT Σ.χρωμα AS χρ
FROM ΙΣ Ι, ΚΡ Κ, ΣΚ Σ
WHERE Ι.κωδι = Κ.κωδι
  AND Σ.κωδσ = Κ.κωδσ
  AND Ι.ονομα = 'Μαρία';

Πράξεις συνόλων — UNION & EXCEPT

9 Σκάφη που κράτησε η Μαρία ή ο Γιώργος — UNION
SELECT DISTINCT Σ.κωδσ
FROM ΣΚ Σ, ΚΡ Κ, ΙΣ Ι
WHERE Ι.κωδι = Κ.κωδι AND Σ.κωδσ = Κ.κωδσ AND Ι.ονομα = 'Μαρία'

UNION

SELECT DISTINCT Σ.κωδσ
FROM ΣΚ Σ, ΚΡ Κ, ΙΣ Ι
WHERE Ι.κωδι = Κ.κωδι AND Σ.κωδσ = Κ.κωδσ AND Ι.ονομα = 'Γιώργος';
10 Όσοι έχουν κράτηση εκτός της Μαρίας — EXCEPT
(SELECT DISTINCT Ι.κωδι
 FROM ΙΣ Ι, ΚΡ Κ
 WHERE Ι.κωδι = Κ.κωδι)

EXCEPT

(SELECT DISTINCT Ι.κωδι
 FROM ΙΣ Ι
 WHERE Ι.ονομα = 'Μαρία');

Υπο-ερωτήματα — IN, NOT IN, ALL

11 Ποιοι κράτησαν το σκάφος 12 — IN
SELECT DISTINCT Ι.ονομα
FROM ΙΣ Ι
WHERE Ι.κωδι IN
    (SELECT Κ.κωδι FROM ΚΡ Κ WHERE Κ.κωδσ = 12);
12 Ποιοι κράτησαν κόκκινο σκάφος — εμφωλευμένο IN
SELECT DISTINCT Ι.ονομα
FROM ΙΣ Ι
WHERE Ι.κωδι IN (
    SELECT Κ.κωδι
    FROM ΚΡ Κ, ΣΚ Σ
    WHERE Κ.κωδσ = Σ.κωδσ AND Σ.χρωμα = 'Κόκκινο'
);
13 Κωδικός μεγαλύτερος από όλους με κόκκινη κράτηση — > ALL
SELECT Ι.ονομα
FROM ΙΣ Ι
WHERE Ι.κωδι > ALL (
    SELECT Κ.κωδι
    FROM ΚΡ Κ, ΣΚ Σ
    WHERE Κ.κωδσ = Σ.κωδσ AND Σ.χρωμα = 'Κόκκινο'
);
14 Ποιοι είχαν κράτηση στις 2026-06-01 — IN με ημερομηνία
SELECT ονομα
FROM ΙΣ
WHERE κωδι IN
    (SELECT κωδι FROM ΚΡ WHERE ημ = '2026-06-01');
15 Σκάφη που δεν έχουν καμία κράτηση — NOT IN
SELECT κωδσ
FROM ΣΚ
WHERE κωδσ NOT IN
    (SELECT κωδσ FROM ΚΡ);

Ύπαρξη — EXISTS & NOT EXISTS

16 Ιστιοπλόοι χωρίς καμία κράτηση — NOT EXISTS
SELECT ονομα
FROM ΙΣ
WHERE NOT EXISTS
    (SELECT * FROM ΚΡ WHERE ΙΣ.κωδι = ΚΡ.κωδι);
17 Ιστιοπλόοι με τουλάχιστον μία κράτηση — EXISTS
SELECT ονομα
FROM ΙΣ
WHERE EXISTS
    (SELECT * FROM ΚΡ WHERE ΙΣ.κωδι = ΚΡ.κωδι);
18 Όσοι κράτησαν αποκλειστικά κόκκινα σκάφη — διπλό NOT EXISTS
SELECT ονομα
FROM ΙΣ
WHERE NOT EXISTS (
    SELECT *
    FROM ΚΡ, ΣΚ
    WHERE ΣΚ.χρωμα != 'Κόκκινο'
      AND ΚΡ.κωδσ = ΣΚ.κωδσ
      AND ΙΣ.κωδι = ΚΡ.κωδι
);
19 Ιστιοπλόοι που ήταν μόνοι τους εκείνη τη μέρα
SELECT ονομα
FROM ΙΣ
WHERE κωδι IN (
    SELECT Κ1.κωδι
    FROM ΚΡ Κ1
    WHERE NOT EXISTS (
        SELECT *
        FROM ΚΡ Κ2
        WHERE Κ1.ημ = Κ2.ημ
          AND Κ1.κωδι != Κ2.κωδι
    )
);
20 Όσοι κράτησαν όλα τα κόκκινα σκάφη — Division
SELECT Ι.ονομα
FROM ΙΣ Ι
WHERE NOT EXISTS (
    SELECT *
    FROM ΣΚ Σ
    WHERE Σ.χρωμα = 'Κόκκινο'
      AND NOT EXISTS (
          SELECT *
          FROM ΚΡ Κ
          WHERE Κ.κωδι = Ι.κωδι
            AND Κ.κωδσ = Σ.κωδσ
      )
);

Συναθροίσεις — COUNT, MIN, MAX, GROUP BY

21 Ιστιοπλόος με τον μικρότερο κωδικό — MIN
SELECT ονομα
FROM ΙΣ
WHERE κωδι = (SELECT MIN(κωδι) FROM ΙΣ);
22 Πόσοι ιστιοπλόοι υπάρχουν συνολικά — COUNT
SELECT COUNT(*) FROM ΙΣ;
23 Πόσες φορές εμφανίζεται κάθε όνομα — GROUP BY
SELECT ονομα, COUNT(*)
FROM ΙΣ
GROUP BY ονομα;
24 Μικρότερος κωδικός που έκανε κράτηση στις 03/06 — MIN + WHERE
SELECT MIN(κωδι)
FROM ΚΡ
WHERE ημ = '2026-06-03';
25 Ιστιοπλόοι με το υψηλότερο επίπεδο — MAX
SELECT ονομα
FROM ΙΣ
WHERE επίπεδο = (SELECT MAX(επίπεδο) FROM ΙΣ);
26 Πλήθος ιστιοπλόων επιπέδου >7 ανά χρώμα — COUNT + GROUP BY
SELECT Σ.χρωμα, COUNT(DISTINCT Ι.κωδι)
FROM ΙΣ Ι, ΣΚ Σ, ΚΡ Κ
WHERE Κ.κωδι = Ι.κωδι
  AND Κ.κωδσ = Σ.κωδσ
  AND Ι.επίπεδο > 7
GROUP BY Σ.χρωμα;

Εργασία 2 — Αεροπορικές Πτήσεις

Πίνακες: airlines, airplanes, airports, routes, flights, passengers, airlines_has_airplanes, flights_has_passengers

Σχήμα Βάσης

Πίνακες 
-- airlines (id, name, code, active)
-- airplanes (id, number, manufacturer)
-- airports (id, name, city, country)
-- routes (id, airlines_id, source_id, destination_id)
-- flights (id, routes_id, airplanes_id, date)
-- passengers (id, name, surname, year_of_birth)
-- airlines_has_airplanes (airlines_id, airplanes_id)
-- flights_has_passengers (flights_id, passengers_id)

Ερωτήματα

1 Αεροσκάφη Airbus που πετούν για Lufthansa
SELECT DISTINCT a.number
FROM airplanes a, airlines_has_airplanes aha, airlines al
WHERE a.manufacturer = 'Airbus'
  AND a.id = aha.airplanes_id
  AND al.id = aha.airlines_id
  AND al.name = 'Lufthansa';
2 Εταιρείες με δρομολόγιο Athens → Prague
SELECT DISTINCT al.name
FROM airlines al, routes r, airports src, airports dest
WHERE al.id = r.airlines_id
  AND r.source_id = src.id AND src.city = 'Athens'
  AND r.destination_id = dest.id AND dest.city = 'Prague';
3 Πόσοι επιβάτες ταξίδεψαν στις 2012-02-19 με Aegean Airlines — COUNT
SELECT COUNT(DISTINCT fhp.passengers_id)
FROM flights_has_passengers fhp, flights f, routes r, airlines al
WHERE fhp.flights_id = f.id
  AND f.date = '2012-02-19'
  AND r.id = f.routes_id
  AND r.airlines_id = al.id
  AND al.name = 'Aegean Airlines';
4 Υπήρξε πτήση Olympic Airways 2014-12-12 Athens→London; (yes/no) — EXISTS + UNION
SELECT 'yes'
WHERE EXISTS (
    SELECT *
    FROM routes r, flights f, airports src, airports dest, airlines al
    WHERE f.date = '2014-12-12'
      AND r.id = f.routes_id
      AND r.airlines_id = al.id AND al.name = 'Olympic Airways'
      AND r.source_id = src.id AND src.name = 'Athens El. Venizelos'
      AND r.destination_id = dest.id AND dest.name = 'London Gatwick'
)

UNION

SELECT 'no'
WHERE NOT EXISTS (
    SELECT *
    FROM routes r, flights f, airports src, airports dest, airlines al
    WHERE f.date = '2014-12-12'
      AND r.id = f.routes_id
      AND r.airlines_id = al.id AND al.name = 'Olympic Airways'
      AND r.source_id = src.id AND src.name = 'Athens El. Venizelos'
      AND r.destination_id = dest.id AND dest.name = 'London Gatwick'
);
5 Μέση ηλικία επισκεπτών Berlin — AVG
SELECT AVG(2026 - p.year_of_birth)
FROM passengers p, flights_has_passengers fhp, flights f, routes r, airports dest
WHERE fhp.passengers_id = p.id
  AND fhp.flights_id = f.id
  AND f.routes_id = r.id
  AND r.destination_id = dest.id
  AND dest.city = 'Berlin';
6 Επιβάτες που πέταξαν πάντα με το ίδιο αεροπλάνο — HAVING COUNT(DISTINCT) = 1
SELECT p.name, p.surname
FROM passengers p, flights f, flights_has_passengers fhp
WHERE fhp.flights_id = f.id
  AND fhp.passengers_id = p.id
GROUP BY p.id, p.name, p.surname
HAVING COUNT(DISTINCT f.airplanes_id) = 1;
7 Πτήσεις 2010-03 έως 2014-07 με πάνω από 5 επιβάτες — BETWEEN + HAVING
SELECT DISTINCT src.city, dest.city
FROM flights f, routes r, airports src, airports dest, flights_has_passengers fhp
WHERE f.routes_id = r.id
  AND r.source_id = src.id
  AND r.destination_id = dest.id
  AND fhp.flights_id = f.id
  AND f.date >= '2010-03-01' AND f.date <= '2014-07-17'
GROUP BY f.id, src.city, dest.city
HAVING COUNT(fhp.passengers_id) > 5;
8 Εταιρείες με ακριβώς 4 αεροσκάφη — αριθμός δρομολογίων — HAVING = 4
SELECT al.id, al.name, al.code, COUNT(DISTINCT r.id)
FROM airlines al, airlines_has_airplanes aha, routes r
WHERE al.id = aha.airlines_id
  AND al.id = r.airlines_id
GROUP BY al.id, al.name, al.code
HAVING COUNT(DISTINCT aha.airplanes_id) = 4;
9 Επιβάτες που πέταξαν με όλες τις ενεργές εταιρείες — Division με COUNT
SELECT p.name, p.surname
FROM passengers p, flights_has_passengers fhp, flights f, routes r, airlines al
WHERE p.id = fhp.passengers_id
  AND fhp.flights_id = f.id
  AND f.routes_id = r.id
  AND r.airlines_id = al.id
  AND al.active = 'Y'
GROUP BY p.id, p.name, p.surname
HAVING COUNT(DISTINCT al.id) = (
    SELECT COUNT(*) FROM airlines WHERE active = 'Y'
);
10 Μόνο Aegean Airlines ή >1 ταξίδι στο 2011–2013 — UNION δύο συνόλων
SELECT p.name, p.surname
FROM passengers p, flights_has_passengers fhp, flights f, routes r, airlines al
WHERE p.id = fhp.passengers_id
  AND fhp.flights_id = f.id
  AND f.routes_id = r.id
  AND r.airlines_id = al.id
GROUP BY p.id, p.name, p.surname
HAVING COUNT(DISTINCT al.name) = 1
   AND MAX(al.name) = 'Aegean Airlines'

UNION

SELECT p.name, p.surname
FROM passengers p, flights_has_passengers fhp, flights f
WHERE p.id = fhp.passengers_id
  AND fhp.flights_id = f.id
  AND f.date BETWEEN '2011-01-02' AND '2013-12-31'
GROUP BY p.id, p.name, p.surname
HAVING COUNT(*) > 1;

Παλιά θέματα

Επισκέπτες & κρατήσεις

Σχετικοί πίνακες: VISITOR (id, surname), KRATISIS (id επισκέπτη, property_id, date), PROPERTY (id, type, …).

1 Επισκέπτες με πάνω από 2 κρατήσεις σε Apartment

Εκφώνηση: Να βρείτε τα ονόματα των επισκεπτών που έχουν κάνει πάνω από 2 κρατήσεις σε ιδιοκτησίες τύπου "Apartment". Να επιστρέφεται και ο συνολικός αριθμός κρατήσεων ανά επισκέπτη.

Λύση: ζεύξη των τριών πινάκων, φίλτρο type = 'Apartment', ομαδοποίηση ανά επισκέπτη, HAVING COUNT(*) > 2.

SELECT v.surname,
       COUNT(*) AS plithos_kratiseon
FROM VISITOR v,
     KRATISIS k,
     PROPERTY p
WHERE v.id = k.id
  AND k.property_id = p.id
  AND p.type = 'Apartment'
GROUP BY v.id, v.surname
HAVING COUNT(*) > 2;
2 Εναλλακτικό — Πάνω από 2 διαφορετικές ιδιοκτησίες

Εκφώνηση: Να βρείτε τα ονόματα των επισκεπτών που έχουν κάνει κρατήσεις σε περισσότερες από 2 διαφορετικές ιδιοκτησίες.

Λύση: COUNT(DISTINCT property_id) ανά επισκέπτη.

FROM VISITOR v,
     KRATISIS k
WHERE v.id = k.id
GROUP BY v.id, v.surname
HAVING COUNT(DISTINCT k.property_id) > 2;
3 Εναλλακτικό — Μόνο σε Apartment

Εκφώνηση: Να βρείτε τα ονόματα των επισκεπτών που έχουν κάνει κρατήσεις μόνο σε ιδιοκτησίες τύπου "Apartment".

Λύση: τουλάχιστον μία κράτηση σε Apartment και καμία κράτηση σε άλλο τύπο — EXISTS + NOT EXISTS.

SELECT v.surname
FROM VISITOR v
WHERE EXISTS (
    SELECT *
    FROM KRATISIS k, PROPERTY p
    WHERE k.id = v.id
      AND k.property_id = p.id
      AND p.type = 'Apartment'
)
AND NOT EXISTS (
    SELECT *
    FROM KRATISIS k, PROPERTY p
    WHERE k.id = v.id
      AND k.property_id = p.id
      AND p.type <> 'Apartment'
);
4 Εναλλακτικό — Πάνω από 2 κρατήσεις στο ίδιο έτος

Εκφώνηση: Να βρείτε τα ονόματα των επισκεπτών που έχουν πραγματοποιήσει περισσότερες από 2 κρατήσεις μέσα στο ίδιο έτος.

Λύση: ομαδοποίηση ανά επισκέπτη και έτος (YEAR(k.date)), HAVING COUNT(*) > 2.

SELECT DISTINCT v.surname
FROM VISITOR v,
     KRATISIS k
WHERE v.id = k.id

Εκφώνηση: Να βρείτε τις ιδιοκτησίες που έχουν δεχτεί περισσότερους από 5 διαφορετικούς επισκέπτες.

Λύση: ομαδοποίηση ανά ιδιοκτησία, COUNT(DISTINCT k.id) για μοναδικούς επισκέπτες.

SELECT p.id,
       COUNT(DISTINCT k.id) AS plithos_episkepton

Βασικές μέθοδοι & λειτουργίες (cursor)

Τι είναι το cursor; Μετά τη σύνδεση με τη βάση (connection = pymysql.connect(...)), δημιουργείς ένα cursor — αντικείμενο που «τρέχει» τα SQL ερωτήματα και φέρνει τα αποτελέσματα:

cursor = connection.cursor()  # Δημιουργία cursor από την ενεργή σύνδεση

Όλες οι μέθοδοι παρακάτω καλούνται στο cursor (όχι απευθείας στη σύνδεση). Η σύνδεση κρατά το commit() / rollback() για τη δοσοληψία.

  • execute() — εκτελεί την εντολή SQL. Επιστρέφει iterator· οι γραμμές έρχονται ως Python tuples.
  • fetchone() — φέρνει την επόμενη γραμμή (row) από το αποτέλεσμα του query.
  • fetchall() — φέρνει όλες τις υπόλοιπες γραμμές από το αποτέλεσμα του query.
  • rowcount — read-only ιδιότητα· αριθμός γραμμών που επηρεάστηκαν από την execute().
  • commit() — οριστικοποιεί τις αλλαγές στη βάση (απαραίτητο σε INSERT / UPDATE).
  • rollback() — ακυρώνει τις αλλαγές σε περίπτωση σφάλματος (συνήθως στο except block).

2. Ροή δεδομένων (Data Flow / Architecture)

Κύκλος ζωής ενός αιτήματος:

  1. Ο περιηγητής στέλνει HTTP Request (GET ή POST) μέσω HTML φόρμας
  2. Το Bottle δρομολογεί (route) στο αντίστοιχο handler Python
  3. Η Python εκτελεί SQL query στη MySQL μέσω PyMySQL
  4. Τα δεδομένα επιστρέφουν και δημιουργείται HTML Response για τον χρήστη

3. Δοσοληψίες (Transactions) και ACID

Οι δοσοληψίες εξασφαλίζουν συνέπεια δεδομένων. Οι 4 ιδιότητες ACID:

  • Ατομικότητα (Atomicity): είτε ολοκληρώνεται πλήρως η δοσοληψία, είτε δεν εφαρμόζεται τίποτα
  • Συνέπεια (Consistency): ξεκινά και τελειώνει με συνεπή κατάσταση της βάσης
  • Απομόνωση (Isolation): ενδιάμεσα αποτελέσματα μη ορατά έξω από την τρέχουσα δοσοληψία
  • Ανθεκτικότητα (Durability): μετά το commit, οι αλλαγές είναι μόνιμες ακόμα και με βλάβη συστήματος

2 Ανάγνωση πινάκου — SELECT & fetchall()

Διαβάζεις γραμμές με fetchall() και τις επεξεργάζεσαι σε βρόχο. Στην εργασία χρησιμοποίησε parameterized queries (βήμα 8), όχι % formatting όπως στο παρακάτω εκπαιδευτικό παράδειγμα.

import pymysql  # Εισαγωγή PyMySQL

db = get_connection()  # Σύνδεση στη βάση (συνήθως από db.py)
cursor = db.cursor()  # Cursor για queries

sql = "SELECT * FROM EMPLOYEE WHERE INCOME > '%d'" % (1000)  # Κατασκευή SQL (εκπαιδευτικό — στην εργασία: parameters)
try:  # Έναρξη χειρισμού σφαλμάτων
    cursor.execute(sql)  # Εκτέλεση SELECT
    results = cursor.fetchall()  # Όλες οι γραμμές αποτελέσματος
    for row in results:  # Επανάληψη για κάθε εγγραφή
        fname = row[0]  # 1η στήλη — όνομα
        lname = row[1]  # 2η στήλη — επώνυμο
        age = row[2]  # 3η στήλη — ηλικία
        sex = row[3]  # 4η στήλη — φύλο
        income = row[4]  # 5η στήλη — εισόδημα
        print("fname = %s, lname = %s, age = %d, sex = %s, income = %d" %  # Εκτύπωση πεδίων
              (fname, lname, age, sex, income))  # Τιμές για το print
except:  # Αν αποτύχει το query
    print("Error: unable to fetch data")  # Μήνυμα σφάλματος

db.close()  # Κλείσιμο σύνδεσης
3 Ενημέρωση — UPDATE, rollback & commit

Οι αλλαγές στη βάση γίνονται μόνιμες με commit(). Σε σφάλμα, rollback() (σχέση με ACID — ατομικότητα).

#!/usr/bin/python3  # Διεύθυνση εκτελέσιμου Python 3
import pymysql  # Βιβλιοθήκη MySQL

db = get_connection()  # Σύνδεση στη βάση
cursor = db.cursor()  # Cursor για εντολές SQL

sql = "UPDATE EMPLOYEE SET AGE = AGE + 1 WHERE SEX = '%c'" % ('M')  # SQL ενημέρωσης ηλικίας (+1 για άνδρες)
try:  # Προσπάθεια εκτέλεσης UPDATE
    cursor.execute(sql)  # Εκτέλεση εντολής
except:  # Σε σφάλμα κατά την execute
    db.rollback()  # Αναίρεση μη οριστικοποιημένων αλλαγών

try:  # Οριστικοποίηση αλλαγών
    db.commit()  # Αποθήκευση μόνιμα στη βάση (Durability μετά από commit)
except:  # Αν αποτύχει το commit
    db.rollback()  # Αναίρεση

db.close()  # Κλείσιμο σύνδεσης

8 Πλήρης ροή — φόρμα → βάση → template

Συνδυασμός όλων των προηγούμενων: POST, ασφαλές SQL με %s και tuple, αποτέλεσμα στο template. Χρειάζεται DictCursor για πρόσβαση με όνομα στήλης.

from bottle import route, request, template, run  # Web + φόρμα + templates
import pymysql.cursors  # DictCursor για αποτελέσματα ως λεξικό

@route('/check_property', method='POST')  # Μόνο POST από φόρμα ελέγχου
def check_property():  # Έλεγχος αν υπάρχει ακίνητο στη βάση
    location = request.forms.get('location')  # Τοποθεσία από τη φόρμα
    property_type = request.forms.get('property_type')  # Τύπος ακινήτου από τη φόρμα

    conn = get_connection()  # Σύνδεση MySQL
    with conn.cursor(pymysql.cursors.DictCursor) as cur:  # Cursor με κλειδιά ονομάτων στηλών
        # Ασφαλές query: placeholders %s, πίνακας Property, exists_flag = 1 αν υπάρχει εγγραφή
        sql = """
            SELECT COUNT(*) > 0 AS exists_flag
            FROM Property
            WHERE location = %s AND property_type = %s
        """
        cur.execute(sql, (location, property_type))  # Εκτέλεση με tuple παραμέτρων (anti SQL injection)
        result = cur.fetchone()  # Μία γραμμή: {"exists_flag": 0 ή 1}
    conn.close()  # Κλείσιμο σύνδεσης

    answer = "yes" if result["exists_flag"] else "no"  # Μεταβλητή για το template
    return template('result', answer=answer)  # Render templates/result.tpl με answer=yes/no
9 Template — templates/result.tpl

Το Bottle αντικαθιστά {{answer}} και εκτελεί λογική με % if / % else / % end.

<!DOCTYPE html> <!-- Δήλωση τύπου εγγράφου HTML5 -->
<html> <!-- Ρίζα σελίδας -->
<head> <!-- Κεφαλίδα (metadata) -->

6. Αυστηροί κανόνες αξιολόγησης

Απόλυτος κανόνας ασφαλείας

  • αλλαγές στη βάση χωρίς commit
  • queries χωρίς parameters
  • αποκλειστικά στατική HTML (χωρίς templates / δυναμικό περιεχόμενο)

Αυστηροί περιορισμοί SQL

  • μόνο εντολές που έχουν διδαχθεί στο μάθημα
  • απαγορεύεται JOIN (LEFT, CROSS, INNER, …)
  • απαγορεύεται LIMIT
  • απαγορεύονται υποερωτήματα SELECT μέσα σε SELECT / FROM
